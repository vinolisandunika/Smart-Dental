  <script>

    const pages = ['home', 'detail', 'video', 'settings'];

    const taskNames = [

      '診察室の準備',

      '器具の滅菌・消毒',

      'バキュームの操作確認'

    ];


    let completed = [false, false, false];


    // Today's date

    document.getElementById('today').textContent =

      new Intl.DateTimeFormat('ja-JP', {

        year: 'numeric', month: '2-digit', day: '2-digit'

      }).format(new Date());


    function showPage(page) {

      pages.forEach(name => {

        document.getElementById(name).classList.toggle(

          'hidden', name !== page

        );

      });


      ['home', 'detail', 'video'].forEach(name => {

        document.getElementById('nav-' + name)

          .classList.toggle('active', name === page);

      });


      window.scrollTo(0, 0);

    }


    function updateTask(index) {

      completed[index] =

        document.getElementById('task-' + index).checked;


      renderTasks();

    }


    function completeTask(index) {

      completed[index] = true;

      renderTasks();

      showPage('home');

    }


    function renderTasks() {

      completed.forEach((done, index) => {

        document.getElementById('task-' + index).checked = done;

        document.getElementById('row-' + index)

          .classList.toggle('done', done);

      });


      const count = completed.filter(Boolean).length;

      document.getElementById('progress').textContent =

        `完了 ${count} / ${completed.length}`;

    }


    function resetTasks() {

      completed = [false, false, false];

      renderTasks();

      showPage('home');

    }


    function showSettings() {

      showPage('settings');

    }


    renderTasks();
